// a934301b
// @draft
// @keywords: objects count select lookup rank
// @scene blocks: on a params.background field stand solid rectangular blocks
//  of cells of other colors; no two blocks touch
// @scene connect: a block is a 4-connected piece of its cells
// @scene mark-color: the mark color is params.mark
// @scene body-and-marks: each block is one body color with some cells of the
//  mark color scattered inside it; a block may hold no mark cell
// @rule mark-count: a block's mark count is its number of mark cells
// @rule keep-few-marks: a block whose mark count is at most params.max_marks is
//  copied to the output unchanged
// @rule drop-many-marks: a block whose mark count is greater than
//  params.max_marks becomes params.background
// @rule background: every other cell is params.background

// how the cells of a block connect
.type Adjacency =
    // @claim connect
    EdgeConnect {}
    // @scene connect: a block is an 8-connected piece of its cells
    // @witness diagonal-touch: borderline concept
    //     @relaxed blocks
  | DiagConnect {}

// which cells form blocks
.type Pieces =
    // @claim blocks
    AnyInk {}
    // @scene blocks: on a params.background field stand solid rectangular
    //  blocks of params.body cells and mark-color cells; no two blocks touch
    // @witness lone-other-color: borderline concept
    //     @relaxed body-and-marks
    // @witness lone-other-fewest: borderline concept
    //     @relaxed body-and-marks
    // @witness other-body-color: variant concept
    // @witness swapped-roles: variant $GridMinority
    //     @relaxed body-and-marks
  | BodyMarkOnly {other: Other}

// what a cell outside the blocks becomes
.type Other =
    // @claim background
    EraseOther {}
    // @rule background: every other cell keeps its input color
    // @witness other-color-on-dropped: borderline $CopyOther
    //     @relaxed body-and-marks
    // @witness other-body-color: variant concept
  | CopyOther {}

// which cells of a block are marks
.type MarkColor =
    // @claim mark-color body-and-marks
    StoredMark {}
    // @scene mark-color: the mark color is the rarer of the grid's two colors
    //  other than params.background; undefined where the grid holds other than
    //  two such colors, or where the two tie
    // @witness mark-majority-grid: borderline concept
    //     @relaxed body-and-marks
    // @witness grid-majority-small: borderline concept
    //     @relaxed body-and-marks
    // @witness grid-majority-pieces: borderline concept
    //     @relaxed body-and-marks
    // @witness rare-background: borderline concept
    //     @relaxed body-and-marks
    // @witness rare-background-small: variant concept
    // @witness third-color-rarest: borderline concept
    //     @relaxed body-and-marks
    // @witness swapped-roles: variant $GridMinority
  | GridMinority {}
    // @scene body-and-marks: a block's mark cells are its cells that are not
    //  of its most frequent color; where two colors tie for the most frequent,
    //  the mark count is the one each of them gives, and undefined where they
    //  give different counts
    // @witness mark-majority-block: borderline concept
    //     @relaxed body-and-marks
    // @witness minority-cells: borderline concept
    //     @relaxed body-and-marks
    // @witness minority-pieces: borderline concept
    //     @relaxed body-and-marks
  | BlockMinority {}

// what is counted of a block's marks
.type Measure =
    // @claim mark-count
    MarkCells {}
    // @rule mark-count: a block's mark count is its number of 4-connected
    //  pieces of mark cells
    // @witness adjacent-marks: ordinary concept
    // @witness clustered-marks: ordinary concept
  | MarkPieces {}

// which mark counts keep a block
.type Keep =
    // @claim keep-few-marks drop-many-marks
    MaxMarks {}
    // @rule keep-few-marks: a block whose mark count is in params.kept_count is
    //  copied to the output unchanged
    // @rule drop-many-marks: a block whose mark count is in
    //  params.dropped_count becomes params.background; undefined where a count
    //  is in neither table
    // @witness non-monotone-table: ordinary $CountTable
    //     @reassigned kept_count
    //     @reassigned dropped_count
    // @witness five-marks: ordinary concept
  | CountTable {}
    // @rule keep-few-marks: a block whose mark count is below the mean mark
    //  count of the grid's blocks is copied to the output unchanged
    // @rule drop-many-marks: every other block becomes params.background
    // @witness high-mean: ordinary concept
  | BelowMean {}
    // @rule keep-few-marks: a block whose mark count is below the median mark
    //  count of the grid's blocks is copied to the output unchanged
    // @rule drop-many-marks: every other block becomes params.background
    // @witness high-median: ordinary concept
  | BelowMedian {}
    // @rule keep-few-marks: the params.keep_count blocks with the fewest marks
    //  are copied to the output unchanged; undefined where the last of them
    //  ties with the next
    // @rule drop-many-marks: every other block becomes params.background
    // @witness two-few-mark-blocks: variant concept
  | FewestThree {}

.type Choice = [adj: Adjacency, pieces: Pieces, mark: MarkColor, measure: Measure, keep: Keep]

// The readings a block's cells depend on.
.type PieceView = [adj: Adjacency, pieces: Pieces, mark: MarkColor]

// The readings a block's mark count depends on.
.type CountView = [v: PieceView, measure: Measure]
.type Counted = [cv: CountView, id: Ident]

.decl out(p: Position, col: Color, ch: Choice)

// The members of each key whose score lies below the mean score of the
// key's members.
.comp MeanSplit<Key, Val> {
    .decl cand(key: Key, v: Val, s: number)  // @input

    .decl below(key: Key, v: Val)
    below(Ky, V) :-
        cand(Ky, V, S),
        N = count : { cand(Ky, _, _) }, T = sum S2 : { cand(Ky, _, S2) },
        S * N < T.
}

.init params = Params

.decl adjacency_choice(a: Adjacency)
adjacency_choice($EdgeConnect()). adjacency_choice($DiagConnect()).

.decl other_choice(o: Other)
other_choice($EraseOther()). other_choice($CopyOther()).

.decl pieces_choice(pc: Pieces)
pieces_choice($AnyInk()).
pieces_choice($BodyMarkOnly(O)) :- other_choice(O).

.decl mark_color_choice(mc: MarkColor)
mark_color_choice($StoredMark()). mark_color_choice($GridMinority()). mark_color_choice($BlockMinority()).

.decl measure_choice(ms: Measure)
measure_choice($MarkCells()). measure_choice($MarkPieces()).

.decl keep_choice(k: Keep)
keep_choice($MaxMarks()). keep_choice($CountTable()). keep_choice($BelowMean()).
keep_choice($BelowMedian()). keep_choice($FewestThree()).

.decl impl_choice(ch: Choice)
impl_choice([A, Pc, Mc, Ms, K]) :-
    adjacency_choice(A), pieces_choice(Pc), mark_color_choice(Mc),
    measure_choice(Ms), keep_choice(K).

.comp Parse<Key> {
    .type KeyColor = [k: Key, col: Color]
    .type View = [k: Key, v: PieceView]
    .type Block = [k: Key, v: PieceView, id: Ident]

    // A block read against one reference color: its mark color, or under
    // BlockMinority one of its most frequent colors.
    .type Reading = [b: Block, ref: Color]

    .type Count = [k: Key, cv: CountView, id: Ident]

    .decl cell(k: Key, p: Position, col: Color)

    // The mark color under each MarkColor branch that has one.
    .decl mark_color(k: Key, mc: MarkColor, col: Color)

    // --- $StoredMark ---
    mark_color(K, $StoredMark(), Col) :- cell(K, _, _), params.mark(Col).

    // --- $GridMinority ---
    // @claim mark-color
    .init grid_colors = Tally<KeyColor, Position>
    grid_colors.item([K, Col], P) :- cell(K, P, Col), !params.background(Col).

    // @claim mark-color
    .init grid_rank = ArgExtremes<Key, Color>
    grid_rank.cand(K, Col, N) :- grid_colors.tally([K, Col], N).

    // @claim mark-color
    .init rarest = Unanimous<Key, Color>
    rarest.item(K, Col) :- grid_rank.low(K, Col, _).

    // The colors other than the rarest: one when the grid holds two colors.
    // @claim mark-color
    .init rest_colors = Unanimous<Key, Color>
    rest_colors.item(K, Col) :- grid_colors.tally([K, Col], _), !grid_rank.low(K, Col, _).

    mark_color(K, $GridMinority(), Col) :- rarest.value(K, Col), rest_colors.value(K, _).

    // The cells that form blocks.
    .decl ink(v: View, p: Position)

    // --- $AnyInk ---
    ink([K, [A, $AnyInk(), Mc]], P) :-
        adjacency_choice(A), mark_color_choice(Mc), cell(K, P, Col), !params.background(Col).

    // --- $BodyMarkOnly ---
    ink([K, [A, $BodyMarkOnly(O), Mc]], P) :-
        adjacency_choice(A), other_choice(O), mark_color_choice(Mc), cell(K, P, Col),
        ( params.body(Col) ; mark_color(K, Mc, Col) ).

    // --- $BodyMarkOnly & $BlockMinority ---
    // Under BlockMinority, params.mark is the other color that forms blocks.
    ink([K, [A, $BodyMarkOnly(O), $BlockMinority()]], P) :-
        adjacency_choice(A), other_choice(O), cell(K, P, Col), params.mark(Col).

    // The cell's block.
    .decl piece(b: Block, p: Position)

    // --- $EdgeConnect ---
    // @claim blocks connect
    .init blocks4 = Components<View>
    blocks4.node([K, [$EdgeConnect(), Pc, Mc]], P) :- ink([K, [$EdgeConnect(), Pc, Mc]], P).

    piece([K, V, Id], P) :- blocks4.label([K, V], P, Id).

    // --- $DiagConnect ---
    // @claim blocks connect
    .init blocks8 = Components8<View>
    blocks8.node([K, [$DiagConnect(), Pc, Mc]], P) :- ink([K, [$DiagConnect(), Pc, Mc]], P).

    piece([K, V, Id], P) :- blocks8.label([K, V], P, Id).

    // Each reading of a block, and its mark cells under that reading.
    .decl reading(r: Reading)
    .decl mark(r: Reading, p: Position)

    // --- $StoredMark, $GridMinority ---
    reading([[K, [A, Pc, Mc], Id], Col]) :- piece([K, [A, Pc, Mc], Id], _), mark_color(K, Mc, Col).
    mark([[K, [A, Pc, Mc], Id], Col], P) :-
        piece([K, [A, Pc, Mc], Id], P), cell(K, P, Col), mark_color(K, Mc, Col).

    // --- $BlockMinority ---
    // @claim body-and-marks
    .init block_colors = Mode<Block, Color, Position>
    block_colors.item([K, [A, Pc, $BlockMinority()], Id], Col, P) :-
        piece([K, [A, Pc, $BlockMinority()], Id], P), cell(K, P, Col).

    reading([X, Body]) :- block_colors.mode(X, Body).
    mark([[K, V, Id], Body], P) :-
        block_colors.mode([K, V, Id], Body), piece([K, V, Id], P), cell(K, P, Col), Col != Body.

    // The mark count of a block under each reading.
    .decl per_reading(c: Count, n: number)

    // --- $MarkCells ---
    // @claim mark-count
    .init mark_cells = Tally<Reading, Position>
    mark_cells.item(R, P) :- mark(R, P).
    mark_cells.domain(R) :- reading(R).

    per_reading([K, [V, $MarkCells()], Id], N) :- mark_cells.tally([[K, V, Id], _], N).

    // --- $MarkPieces ---
    // @claim mark-count
    .init mark_blobs = Components<Reading>
    mark_blobs.node(R, P) :- mark(R, P).

    // @claim mark-count
    .init mark_pieces = Tally<Reading, Ident>
    mark_pieces.item(R, Id) :- mark_blobs.label(R, _, Id).
    mark_pieces.domain(R) :- reading(R).

    per_reading([K, [V, $MarkPieces()], Id], N) :- mark_pieces.tally([[K, V, Id], _], N).

    // A block's mark count: the count every reading of it gives.
    // @claim mark-count
    .init marks = Unanimous<Count, number>
    marks.item(C, N) :- per_reading(C, N).
}

.init parse = Parse<Unit>
parse.cell($U(), P, Col) :- in(P, Col).

// A block's mark count.
.decl marks(c: Counted, n: number)
marks([Cv, Id], N) :- parse.marks.value([$U(), Cv, Id], N).

// The blocks each keep reading keeps, and those it drops.
.decl kept(c: Counted, k: Keep)
.decl dropped(c: Counted, k: Keep)

// What a cell outside the blocks becomes.
.decl rest(pc: Pieces, p: Position, col: Color)

out(P, Col, [A, Pc, Mc, Ms, K]) :-
    parse.piece([$U(), [A, Pc, Mc], Id], P),
    ( kept([[[A, Pc, Mc], Ms], Id], K), in(P, Col)                      // kept block: copied
    ; dropped([[[A, Pc, Mc], Ms], Id], K), params.background(Col)       // dropped block: erased
    ).

out(P, Col, [A, Pc, Mc, Ms, K]) :-
    impl_choice([A, Pc, Mc, Ms, K]), rest(Pc, P, Col), !parse.ink([$U(), [A, Pc, Mc]], P).

// --- $AnyInk, $EraseOther ---
rest($AnyInk(), P, Col) :- in(P, _), params.background(Col).
rest($BodyMarkOnly($EraseOther()), P, Col) :- in(P, _), params.background(Col).

// --- $CopyOther ---
rest($BodyMarkOnly($CopyOther()), P, Col) :- in(P, Col).

// --- $MaxMarks ---
kept(C, $MaxMarks()) :- marks(C, N), params.max_marks(M), N <= M.
dropped(C, $MaxMarks()) :- marks(C, N), params.max_marks(M), N > M.

// --- $CountTable ---
kept(C, $CountTable()) :- marks(C, N), params.kept_count(N).
dropped(C, $CountTable()) :- marks(C, N), params.dropped_count(N).

// --- $BelowMean ---
// @claim keep-few-marks drop-many-marks
.init mean_test = MeanSplit<CountView, Ident>
mean_test.cand(Cv, Id, N) :- marks([Cv, Id], N).

kept([Cv, Id], $BelowMean()) :- mean_test.below(Cv, Id).
dropped([Cv, Id], $BelowMean()) :- marks([Cv, Id], _), !mean_test.below(Cv, Id).

// --- $BelowMedian, $FewestThree ---
// The blocks of X's grid with at most X's mark count, X included.
.decl at_most(c: Counted, j: Ident)
at_most([Cv, X], J) :- marks([Cv, X], N), marks([Cv, J], M), M <= N.

// --- $BelowMedian ---
// The blocks of X's grid with more marks than X.
.decl above(c: Counted, j: Ident)
above([Cv, X], J) :- marks([Cv, X], N), marks([Cv, J], M), M > N.

// X's count is below the median unless the blocks with at most X's count
// outnumber those with more.
// @claim keep-few-marks drop-many-marks
.init median_test = Excess<Counted, Ident>
median_test.plus(C, J) :- at_most(C, J).
median_test.minus(C, J) :- above(C, J).

kept(C, $BelowMedian()) :- marks(C, _), !median_test.excess(C, _).
dropped(C, $BelowMedian()) :- median_test.excess(C, _).

// --- $FewestThree ---
// @claim keep-few-marks
.init upto_rank = Tally<Counted, Ident>
upto_rank.item(C, J) :- at_most(C, J).

// The blocks with fewer marks than X.
// @claim drop-many-marks
.init fewer_rank = Tally<Counted, Ident>
fewer_rank.item([Cv, X], J) :- marks([Cv, X], N), marks([Cv, J], M), M < N.

kept(C, $FewestThree()) :- upto_rank.tally(C, N), params.keep_count(K), N <= K.
dropped(C, $FewestThree()) :- fewer_rank.tally(C, N), params.keep_count(K), N >= K.

.comp Determine {
    .init parse = Parse<Example>
    parse.cell(E, P, Col) :- ex_in(E, P, Col).

    // The mark count of each block the concept reads.
    .decl mark_count(e: Example, id: Ident, n: number)
    mark_count(E, Id, N) :-
        parse.marks.value([E, [[$EdgeConnect(), $AnyInk(), $StoredMark()], $MarkCells()], Id], N).

    // A block a train output erases: some cell of it is background there.
    .decl erased(e: Example, id: Ident)
    erased(E, Id) :-
        parse.piece([E, [$EdgeConnect(), $AnyInk(), $StoredMark()], Id], P),
        ex_out(E, P, Col), params.background(Col).

    .decl kept_count(n: number)
    kept_count(N) :- mark_count(E, Id, N), ex_out(E, _, _), !erased(E, Id).

    .decl dropped_count(n: number)
    dropped_count(N) :- mark_count(E, Id, N), erased(E, Id).
}
